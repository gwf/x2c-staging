 (rule number
  (seq (optional (one-of-bytes "+-"))
   (select-radix
    (ordered
     (when-prefix (seq (byte 48) (one-of-bytes "xX"))
      (choice (ref hex-float) (ref hex-integer)))
     (when-prefix (seq (byte 48) (one-of-bytes "bB")) (ref binary-integer))
     (when-prefix (seq (byte 48) (one-of-bytes "oO")) (ref octal-integer))
     (when (and (prefix (seq (byte 48) (range 49 55)))
            (not (after-decimal-digits (one-of-bytes ".eE"))))
      (ref legacy-octal))
     (otherwise (choice (ref decimal-float) (ref decimal-integer))))))

predicates

(predicates
 (predicate percent-is-operator
  (and (not (predicate layout-statement-start))
   (previous-kind-in operand-end) (not (predicate closes-control))
   (not (predicate closes-statement-block))))
 (predicate can-start-symbol
  (or (predicate layout-statement-start) (not (previous-kind-in operand-end))
   (previous-ident "tag")
   (previous-sequence (kind-in operand-end)
    (ident "is") (optional (ident "not")))))
 (predicate layout-statement-start
  (and layout-enabled (last-stored-kind space)
   (last-stored-text-contains (byte 10)) (previous-significant-exists)
   (column-at-most first-significant-on-previous-token-physical-line)))
 (predicate closes-control
  (and (previous-kind ")")
   (before-matching-ordinary-group-kind-in "if"
    "while" "for" "switch")))
 (predicate closes-statement-block
  (and (previous-kind "}")
   (or (before-matching-ordinary-group-kind-in
        ";" ":" "{" "}" "]" "ident" "else" "do" "try"
        "finally" "defer")
    (before-matching-ordinary-group (kind ")")
     (before-matching-ordinary-group-kind-in "ident"
      "match" "if" "while" "for" "switch")))))
 (predicate lisp-angle-start
  (and (prefix (byte 60))
   (not (next-byte-in (byte 0) (byte 61) (ref white-byte)
         (one-of-bytes "()\'`,")))))
