Extend the language.

Operators, macros, and compile-time code that shape the language.

compile time

Compute while the program compiles.

A meta function runs x2c during compilation. Build related tables from one description, then use the generated values in the finished program.

examples/magic/compile-time.xlink
#include <assert.h>

static const int codes[] = $column(0);

// The compiler runs `column` and inserts a C initializer built from the table.
meta static List column(int n) {
  Array cells = [];
  foreach (List row in responses()) cells.push(row[n]);
  List items = cells.list_free();
  return $!( { @items } );
}

// One table supplies both wire codes and readable labels.
meta static List responses(void) =>
  %((200 "OK") (404 "Not Found") (503 "Unavailable"));

int main(void) {
  String labels[] = $column(1);

  // The generated program uses ordinary C arrays.
  for (size_t i = 0; i < sizeof(codes) / sizeof(*codes); i++)
    printf("%d: %s\n", codes[i], labels[i]);
  assert(codes[1] == 404 && labels[1] == "Not Found");
  return 0;
}

Actual output

Expected output
200: OK
404: Not Found
503: Unavailable

Run locally: x2c run examples/magic/compile-time.x
macros
Make a repeated pattern a construct.


A macro receives syntax and returns syntax. This swap works with different types because it expands into the surrounding program, where the compiler resolves those types.


examples/magic/macros.xlink
#include <assert.h>
macro Stmt $swap(Expr $left, Expr $right) {
  $Code.type($left) temporary = $left;
  $left = $right;
  $right = temporary;
}

int main(void) {
  // The same macro works with different types.
  int left = 20, right = 22;
  $swap(left, right);

  String first = "hello", last = "world";
  $swap(first, last);
  printf("%d %d; %s %s\n", left, right, first, last);
  assert(left == 22 && right == 20);
  assert(first == "world" && last == "hello");
  return 0;
}
Actual output
Expected output
22 20; world hello

Run locally: x2c run examples/magic/macros.x
decorators
Transform a function at its definition.


A decorator adds tracing around a function body. The function keeps its ordinary calling interface while the transformation supplies the repeated entry and exit work.


examples/magic/decorators.xlink
#include <assert.h>
macro Decorator $trace(Function $function) {
  printf("enter %s\n", $Code.name($function));
  defer printf("leave %s\n", $Code.name($function));
  @Code.body($function)
}

$trace()
int answer(void) => 42;

int main(void) {
  // Prints "enter answer", then "leave answer".
  int result = answer();
  assert(result == 42);
  return 0;
}
Actual output
Expected output
enter answer
leave answer

keywords
Give a convention its own syntax.


Turn locking and cleanup into a statement. The decorator introduces the held lock and its deferred release, so the body can concentrate on the operation being protected.


examples/magic/keywords.xlink
#include <assert.h>
macro Decorator $control.with_lock(Stmt $body, Expr $lock) { {
    Mutex held = $lock;
    held.lock();
    defer held.unlock();
    $body
  } }

keyword with_lock $control.with_lock;

int main(void) {
  auto Mutex mutex = Mutex.new();
  try with_lock(mutex) { raise %(oops); }
  catch %(oops):
  {
    assert(mutex.try_lock());
    mutex.unlock();
    puts("caught, and the lock is already free");
  }
  return 0;
}
Actual output
Expected output
caught, and the lock is already free

reflection
The compiler can describe your types.


Inspect a type’s fields at compile time and generate code that visits them. This is the same general technique behind derived operations: the declaration supplies the information instead of a second handwritten field list.


examples/magic/fields.xlink
#include "meta.x"

typedef struct Reading { int celsius; String station; int gusts; } Reading;

// The compiler hands a meta function the fields of a struct type.
meta static List show_fields(List receiver, TypeInfo type) {
  Array lines = [];
  foreach (List field in type.assoc(<fields>)) {
    String name = field.car();
    List label = $!String{ $name };
    lines.push($!Stmt{ printf("%s = %s\n", $label, (Var) $receiver.$name); });
  }
  return $!Stmt{ { @(lines.list_free()) } };
}

// The macro captures the expression; the meta function writes the code.
macro Stmt $show(Expr $value) => $show_fields($value, $value);

int main(void) {
  Reading noon = { 21, "roof", 12 };
  $show(noon);
  return 0;
}
Actual output
Expected output
celsius = 21
station = roof
gusts = 12

autodiff
Differentiate the program.


Automatic differentiation transforms a function into code that computes derivatives: $ad.reverse() generates energy_grad, which returns the value and writes both partial derivatives. The example combines syntax transformation with ordinary numeric code; the linked full source includes executable checks.


packages/autodiff/examples/energy-gradient.xlink
// Differentiate a loop with a continue in reverse mode, then check the
// gradient against a central finite difference.
#include <assert.h>
#include <math.h>
import "autodiff";

$ad.reverse()
static double energy(double x, double y, int n) {
  double s = 0.0;
  for (int i = 0; i < n; i++) {
    if (i == 2) continue;
    s += x * x * (double) i;
  }
  return sin(s) / x - exp(y);
}

int main(void) {
  double dx, dy;
  double value = energy_grad(1.0, 2.0, 5, &dx, &dy);
  double h = 1e-6;
  double check = (energy(1.0 + h, 2.0, 5) - energy(1.0 - h, 2.0, 5)) / (2 * h);
  assert(fabs(value - energy(1.0, 2.0, 5)) < 1e-12);
  assert(fabs(dx - check) < 1e-5);
  printf("energy %.6f  d/dx %.6f  d/dy %.6f\n", value, dx, dy);
  return 0;
}

runtime Lisp
Keep a rule editable at runtime.


Evaluate a Lisp expression using a native binding supplied by the host program. The surrounding program stays compiled while the policy remains data that can be changed and evaluated.


examples/magic/runtime.xlink
#include <assert.h>

int seats_left(String plan) => plan == "pro" ? 25 : 2;

int main(void) {
  // Expose an ordinary typed function to the runtime interpreter.
  auto Lisp lisp = Lisp.new();
  $lisp.bind(lisp, "seats-left", seats_left);

  // Keep the party together, or admit as many people as fit.
  List together = %(lambda (plan n)
    (if (<= n (seats-left plan)) n 0));
  List split = %(lambda (plan n)
    (if (<= n (seats-left plan)) n (seats-left plan)));

  // Supply either policy as data to the same evaluator.
  int whole = lisp.eval($(together, "free", 12));
  int partial = lisp.eval($(split, "free", 12));
  printf("together: %d; split: %d\n", whole, partial);
  assert(whole == 0 && partial == 2);
  return 0;
}
Actual output
Expected output
together: 0; split: 2

operators


Existing collections do the storage work; the new type supplies the
meaning. Set.new reuses Map storage. Set.add and Set.mul define union
and intersection, and protocol Map(Set) connects them to + and *.
The operators return new sets, leaving both inputs unchanged.


examples/magic/operators.xlink
#include <assert.h>
// Reuse Map storage: its keys are the set members.
typedef Map Set;

Set Set.add(Set left, Set right) => left.copy().merge(right);
Set Set.mul(Set left, Set right) => Set.new(left.keys()
  .filter(%!(Var key) => key in right));
Set Set.new(List keys) => keys.foldl(%{},
  %!(Map set, Var key) => set.update_n(1, key, (Var) 1));

// Connect add and mul to the + and * operators.
protocol Map(T) { T T.add(T, T); T T.mul(T, T); }
protocol Map(Set);

int main(void) {
  // Create two sets, then take their union and intersection.
  Set fruit = Set.new(%(apple pear));
  Set lunch = Set.new(%(pear bread));
  Set set_union = fruit + lunch;
  Set intersection = fruit * lunch;
  printf("union: %s\n", set_union.keys().list().sort().repr());
  printf("intersection: %s\n", intersection.keys().list().sort().repr());
  assert(set_union.len() == 3 && set_union.contains(<apple>) &&
         set_union.contains(<pear>) && set_union.contains(<bread>));
  assert(fruit.len() == 2 && lunch.len() == 2);
  assert(Set.new(nil).len() == 0);
  assert(Set.new(%(apple pear apple)).len() == 2);
  assert(intersection.len() == 1 && intersection.contains(<pear>));
  assert((fruit * fruit).len() == 2);
  assert((fruit * Set.new(nil)).len() == 0);
  assert((Set.new(nil) * fruit).len() == 0);
  assert(fruit.len() == 2 && lunch.len() == 2);
  return 0;
}

Expected output
union: (apple bread pear)
intersection: (pear)

same list


A meta function computes the squares during translation, a literal writes
(1 4 9) out, and direct cons calls build the same cells. Runtime Lisp calculates the squares
on the fly to produce the same List.
All four results compare equal in constant time because they resolve to the
exact same object.


examples/magic/ways-to-agree.xlink
#include <assert.h>

// A meta function computes the List during translation.
meta static List squares(int n) {
  Array items = [];
  for (int i = 1; i <= n; i++) items.push(i * i);
  return items.list_free();
}

int main(void) {
  List comptime = $squares(3);

  // A literal compiles to a form similar to the cons cells below.
  List literal = %(1 4 9);

  // Explicit construction reveals the underlying C functions.
  List constructed = cons(int_var(1),
    cons(int_var(4), cons(int_var(9), NULL)));

  // Runtime Lisp will build it on the fly.
  auto Lisp lisp = Lisp.new();
  List runtime = lisp.eval(%(list (* 1 1) (* 2 2) (* 3 3)));

  List ways = $(comptime, literal, constructed, runtime);
  foreach (List left in ways)
    foreach (List right in ways) assert(left == right);
  assert(comptime == literal && literal == constructed &&
         constructed == runtime);
  puts("All four lists are equal.");
  return 0;
}
Actual output
Expected output
All four lists are equal.
