Compose larger programs.

Types, protocols, control flow, and concurrency, working together.

classes

Your types join in.

Define the data and behavior of a Point or an Account. Constructors, methods, and generated operations let your own types participate in the same style of code as the built-ins.

examples/power/classes.xlink
#include <assert.h>

// A value class: construction, equality, hashing, and printing come free.
class Point { int x; int y; };

// A pointer class with your own init; new allocates in the current Scope.
class Account { String owner; long cents; } *;

void Account.init(Account account, String owner) {
  account.owner = owner;
  account.cents = 0;
}

void Account.deposit(Account account, long cents) => account.cents += cents;
String Account.repr(Account account) =>
  %"Account($(account.owner), $(account.cents) cents)";

int main(void) {
  Point here = Point.new(3, 4), there = Point.new(3, 4);
  assert(here == there);
  Map labels = {};
  labels[here] = "home";
  printf("%s is %s\n", there.repr(), labels[there]);

  Account ada = auto Account.new("Ada");
  ada.deposit(1250);
  ada.deposit(75);
  printf("%s\n", ada.repr());
  return 0;
}

Actual output

Expected output
Point { x: 3, y: 4 } is home
Account(Ada, 1325 cents)

protocols
Small contracts, useful combinations.


A Set built on a Map gains union and intersection through operators. The type is a typedef Map Set: it reuses Map storage and operations while giving the result a vocabulary of its own.


examples/power/protocols.xlink
#include <assert.h>
// Reuse Map storage: its keys are the set members.
typedef Map Set;

Set Set.add(Set left, Set right) => left.copy().merge(right);
Set Set.mul(Set left, Set right) => Set.new(left.keys()
  .filter(%!(Var key) => key in right));
Set Set.new(List keys) => keys.foldl(%{},
  %!(Map set, Var key) => set.update_n(1, key, (Var) 1));

// Connect add and mul to the + and * operators.
protocol Map(T) { T T.add(T, T); T T.mul(T, T); }
protocol Map(Set);

int main(void) {
  // Create two sets, then take their union and intersection.
  Set fruit = Set.new(%(apple pear));
  Set lunch = Set.new(%(pear bread));
  Set set_union = fruit + lunch;
  Set intersection = fruit * lunch;
  assert(set_union.len() == 3 && set_union.contains(<apple>) &&
         set_union.contains(<pear>) && set_union.contains(<bread>));
  assert(fruit.len() == 2 && lunch.len() == 2);
  assert(Set.new(nil).len() == 0);
  assert(Set.new(%(apple pear apple)).len() == 2);
  Set symbols = Set.new(%(apple pear));
  assert(symbols.contains(<apple>) && symbols.contains(<pear>));
  assert(intersection.len() == 1 && intersection.contains(<pear>));
  assert((fruit * fruit).len() == 2);
  assert((fruit * Set.new(nil)).len() == 0);
  assert((Set.new(nil) * fruit).len() == 0);
  assert(fruit.len() == 2 && lunch.len() == 2);
  printf("%d in the union, %d in both\n", set_union.len(), intersection.len());
  return 0;
}
Actual output
Expected output
3 in the union, 1 in both

indexing
Build an index. Ask it questions.


Collect document words into an index, then combine the indexed results. Maps, Arrays, and iterator operations make the useful structure visible in the program.


examples/power/indexing.xlink
#include <assert.h>
// Store documents once; words refer to them by ID.
typedef struct Index { Array documents; Map words; } *Index;

Index Index.new(void) {
  Index index = xmalloc(sizeof(struct Index));
  index.documents = []; index.words = {};
  return index;
}

// Add each word once per document, regardless of case or repetition.
void Index.add(Index index, String name, String text) {
  int id = index.documents.len();
  index.documents.push(name);
  foreach (String word in text.lower().words().iter().unique())
    index.words.setdefault(word, []).array().push(id);
}

// An unknown word has no matches; lookup does not change the index.
Array Index.find(Index index, String word)
  => index.words.getdefault(word.lower(), []);

int main(void) {
  Index index = Index.new();
  index.add("garden", "red roses red tulips");
  index.add("kitchen", "red apples green pears");
  assert(index.find("RED") == [0, 1]);
  assert(index.find("roses") == [0]);
  int words = index.words.len();
  assert(index.find("blue").len() == 0 && index.words.len() == words);
  Array matches = index.find("roses");
  puts(%"roses: $(index.documents[matches[0]])");
  return 0;
}
Actual output
Expected output
roses: garden

matching
Recognize a shape. Rebuild it.


match selects nested List structures and binds their parts. Templates rebuild the result: $ inserts a value and @ splices a sequence.


This small simplifier combines pattern matching and reconstruction without manually indexing every node.


examples/power/match.xlink
#include <assert.h>

// Recognize a nested condition and exchange its branches.
List simplify(List node) {
  match (node) {
    case %(if (not ?condition) ?yes ?no):
      return simplify(%(if $condition $no $yes));
    case %(if true ?yes ?): return simplify(yes);
    case %(if false ? ?no): return simplify(no);
    case %(if ?condition ?yes ?no):
      return %(if $condition $(simplify(yes)) $(simplify(no)));
    // Capture a whole block, transform it, splice it back in.
    case %(block *body):
      return %(block @(body.map(%!(List item) => simplify(item))));
    default: return node;
  }
}

int main(void) {
  List code = %(block
    (if (not ready) (call wait) (call start))
    (if true (call save) (call discard)));
  puts(simplify(code).repr());
  // (block (if ready (call start) (call wait)) (call save))
  assert(simplify(code) ==
         %(block (if ready (call start) (call wait)) (call save)));
  assert(simplify(%(if false (call wrong) (call right))) == %(call right));
  assert(simplify(%(if (not true) (call wrong) (call right))) == %(call right));
  assert(simplify(%(if ready (if true (call yes) (call wrong))
                            (if false (call wrong) (call no)))) ==
         %(if ready (call yes) (call no)));
  assert(simplify(%(block)) == %(block));
  assert(simplify(%(call untouched)) == %(call untouched));
  return 0;
}
Actual output
Expected output
(block (if ready (call start) (call wait)) (call save))

iterators
Connect operations without collecting every step.


Iterators select, merge, and enumerate values as they are consumed. Compose the operations around the task, then collect or reduce the result where you need it.


examples/power/iterators.xlink
/*  iterators.x -- lazy selection, merging, and validation */

int main(void) {
  auto {
    List seen = %(3 1 3 2 1 4);
    printf("unique: %s\n", seen.iter().unique().list().repr());

    List urgent = %(101 102), routine = %(201 202 203);
    printf("queue: %s\n", urgent.iter()
      .chain(routine.iter()).head(4).list().repr());

    printf("first square over threshold: %d\n", range(1, 20)
      .map(%!(value) => value * value)
      .find(%!(value) => value > 70).int());

    List temperatures = %(71 106 98 112 87);
    printf("temperature alerts: %d\n", temperatures.iter()
      .filter(%!(value) => value > 100).count());
    printf("temperature warning: %s\n", temperatures.iter()
      .any(%!(value) => value > 110) ? "yes" : "no");

    List scores = %(91 88 100 73);
    printf("scores valid: %s\n", scores.iter()
      .all(%!(score) => score >= 0 && score <= 100) ? "yes" : "no");

    printf("numbered: %s\n", %(<parse> <compile> <link>).iter()
      .enumerate(1).list().repr());
  }
  return 0;
}

Expected output
unique: (3 1 2 4)
queue: (101 102 201 202)
first square over threshold: 81
temperature alerts: 2
temperature warning: yes
scores valid: yes
numbered: ((1 parse) (2 compile) (3 link))

records
Describe the record once.


Generated record operations keep field mappings, equality, and representation tied to the type declaration. The same fields work with the surrounding collection and serialization operations.


examples/power/derive.xlink
/*  derive.x -- record mapping, representation, and nested equality */
#include <derive.x>

typedef struct Point { int x, y; } Point;
typedef struct Person {
  int age;
  String name;
  double score;
  Point home;
  Array tags;
} Person;

$derive(Point, <map>, <equal>, <repr>);
$derive(Person, <json>, <equal>, <repr>);

int main(void) {
  Person p = { 42, "Ada", 1.5, { 3, 4 }, ["x", 2] };
  Map m = p.to_map();
  Map home = m["home"];
  printf("map age=%s name=%s score=%s home=(%s, %s) tags=%s\n",
         m["age"], m["name"], m["score"], home["x"], home["y"], m["tags"]);
  Person q = Person.from_map(m);
  printf("%s\n", q.repr().str());
  printf("equal %d\n", p.equal(q));
  q.home.y = 9;
  printf("equal %d %s\n", p.equal(q), q.repr().str());
  return 0;
}
Actual output
Expected output

Person{ age: 42, name: "Ada", score: 1.5, home: Point{ x: 3, y: 4 }, tags: [ "x", 2 ] }
equal 1
equal 0 Person{ age: 42, name: "Ada", score: 1.5, home: Point{ x: 3, y: 9 }, tags: [ "x", 2 ] }

errors
Cleanup and recovery belong together.


A failed read can leave a block without leaving its file open. Cleanup stays beside the resource; the caller handles the failure where it can decide what to do next.


examples/power/exceptions.xlink
#include <assert.h>
#include "path.x"
#include <fcntl.h>
#include <errno.h>
static int opened_fd = -1;

// The caller decides how to present content that cannot be text.
String preview(String path) {
  try return read_text(path);
  catch %(bad-arg *):
  {
    assert(fcntl(opened_fd, F_GETFD) == -1 && errno == EBADF);
    return "No text preview: file contains a NUL byte.";
  }
  catch %(not-found *):
    return "No preview: file not found.";
  catch:
    return "No preview: unable to read file.";
}

// Close the file on success or while unwinding from a failed read.
String read_text(String path) {
  auto File input = path.open("r");
  opened_fd = input.fileno();
  return input.string();
}

// The checked-in data sits beside the examples wherever they live.
String example_data(String name) {
  Path source = String.new(__FILE__);
  return source.dirname().join("../data").join(name);
}

int main(int argc, char **argv) {
  String directory = argc > 1 ? argv[1]
                             : example_data("power-exceptions");
  assert(preview(%"$directory/document.txt") == "Meeting at noon.");
  assert(fcntl(opened_fd, F_GETFD) == -1 && errno == EBADF);
  // document.bin contains a NUL byte, which String cannot hold.
  puts(preview(%"$directory/document.bin"));
  assert(fcntl(opened_fd, F_GETFD) == -1 && errno == EBADF);
  assert(preview(%"$directory/missing-preview-file.txt")
         == "No preview: file not found.");
  assert(preview(directory) == "No preview: unable to read file.");
  assert(fcntl(opened_fd, F_GETFD) == -1 && errno == EBADF);
  return 0;
}
Actual output
Expected output
No text preview: file contains a NUL byte.

packages
Bring a library into the language.


Import PCRE2 under the name rx, compile a pattern, and read named captures. Package types and methods work alongside ordinary x2c Strings, iteration, and scoped cleanup.


Full source.


examples/power/imports.xlink
/*  imports.x -- extract named request fields with the PCRE2 package

    In a checkout, `make -C packages/pcre2 run` builds the package and runs
    this example. With an installed x2c, run `x2c install pcre2` first; then
    `x2c run -q examples/power/imports.x` finds the package.
*/

#include <assert.h>
// Import the package under a name you choose.
import "pcre2" as rx;

int main(void) {
  // Compile a pattern with named captures; release it on exit.
  auto rx.Regexp request = rx.Regexp.compile(
    "^(?<method>[A-Z]+) (?<path>/[^ ]+)$", 0);

  // Sample requests include lines the pattern will ignore.
  String text = %"\
GET /docs
POST /build
-- idle --
DELETE /cache
GET missing-slash
";

  int matches = 0;
  foreach (String line in text.lines()) {
    rx.RegexpMatch found = request.match(line);
    if (found)
    {
      puts(%"$(found[<method>]) requests $(found[<path>])");
      matches++;
    }
  }
  assert(matches == 3);
  return 0;
}
This program needs a package, so it runs only with the native toolchain.
Expected output

POST requests /build
DELETE requests /cache

channels
Give each worker a job.


Channels connect a producer and consumer. One worker updates the counts, and the caller reads the result after joining. Ownership and communication remain visible in the code.


examples/power/channels.xlink
#include <assert.h>

int main(void) {
  // A bounded channel carries values between native threads.
  Channel lines = auto Channel.new(<string>, 4);
  Map counts = {};

  // Thread.run starts every worker and joins them all before returning.
  Thread.run([
    %!(void) => {
      foreach (String line in %("tea and cake" "cake and more cake"))
        lines.send(line);
      lines.close();
    },
    %!(void) => {
      foreach (String line in lines)
        foreach (String word in line.words()) counts[word] += 1;
    }
  ]);

  assert(counts["cake"] == 3 && counts["and"] == 2);
  printf("cake: %s, tea: %s\n", counts["cake"], counts["tea"]);
  return 0;
}
This program uses threads, so it runs only with the native toolchain.
Expected output
cake: 3, tea: 1

typed collections
Choose the representation you need.


Typed collections retain concrete element types while offering familiar collection operations. Count response codes without giving up native numeric storage.


examples/power/generics.xlink
#include <assert.h>
// Opt into the ready-made typed collection families.
#include "typed-array.x"
#include "typed-map.x"
#include "typed-list.x"

int main(void) {
  // Store native ints and count each response code.
  ArrayInt responses = [200, 200, 404, 200, 500, 404];
  MapIntInt counts = {};
  for (int i = 0, n = responses.len(); i < n; i++)
    counts[responses[i]] += 1;

  // A typed List fixes the report order, including absent codes.
  ListInt codes = %(200 403 404 500);
  for (ListInt rest = codes; rest; rest = rest.cdr()) {
    int code = rest.car();
    printf("%d: %d\n", code, counts.getdefault(code, 0));
  }
  assert(counts[200] == 3 && counts[404] == 2);
  assert(counts[500] == 1 && counts.getdefault(403, 0) == 0);
  assert(counts.len() == 3 && responses.len() == 6);
  assert(codes.car() == 200 && codes.last() == 500);
  return 0;
}
Actual output
Expected output
200: 3
403: 0
404: 2
500: 1

counting


File iteration, String.lower, and Map updates count words without
manual buffers or table management. Array.sort_with compares the counts
in each (word count) pair to rank the results: 3 tea, 2 coffee, 1 milk.
Ties sort by word in reverse order.


examples/power/counting.xlink
#include "path.x"

// Files supply lines; each line supplies words.
Map word_count(String path) {
  Map counts = {};
  auto File input = path.open("r");
  foreach (String line in input)
    foreach (String word in line.lower().words())
      counts[word] += 1;
  return counts;
}

// The checked-in data sits beside the examples wherever they live.
String example_data(String name) {
  Path source = String.new(__FILE__);
  return source.dirname().join("../data").join(name);
}

int main(int argc, char **argv) {
  // words.txt: "Tea coffee tea" then "Milk coffee tea".
  String path = argc > 1 ? argv[1]
                        : example_data("power-counting/words.txt");
  Map counts = word_count(path);

  // Rank by count, breaking ties by word in the same descending order.
  Array ranked = counts.enumerate().array().sort_with(
    %!(List left, List right) => {
      int order = right[1].compare(left[1]);
      return order ? order : right[0].compare(left[0]);
    });
  foreach ((String word, int n) in ranked)
    puts(%"$n $word");
  return 0;
}
This program reads files from disk, so it runs only with the native toolchain.
Expected output
3 tea
2 coffee
1 milk

threads


Thread.start_raw copies each batch’s bytes and launches a native worker.
Both workers start before either Thread.join returns a result; their
partial sums combine to 91. Thread.free releases the finished handles.
The workers need neither shared input pointers nor a shared accumulator.


examples/power/threads.xlink
#include <assert.h>

// Each worker squares its values and reduces them to one sum.
static Var sum_squares(const void *input, size_t bytes) {
  const int *values = input;
  double sum = 0;
  for (size_t i = 0; i < bytes / sizeof(int); i++)
    sum += (double) values[i] * values[i];
  return sum;
}

int main(void) {
  // Split the data; launch both workers before waiting for either.
  int values[2][3] = {{1, 2, 3}, {4, 5, 6}};
  Thread first = Thread.start_raw(sum_squares,
    values[0], sizeof(values[0]));
  Thread second = Thread.start_raw(sum_squares,
    values[1], sizeof(values[1]));
  values[0][0] = 100;
  values[1][0] = 100;

  // Combine the partial sums after both workers finish.
  double left = first.join(), right = second.join();
  first.free(); second.free();
  printf("sum of squares: %g\n", left + right);
  assert(left == 14 && right == 77 && left + right == 91);
  return 0;
}
This program uses threads, so it runs only with the native toolchain.
Expected output
sum of squares: 91
